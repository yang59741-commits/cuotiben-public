\section*{南开大学2026年数学分析真题}
\section*{57．南开大学2026年数学分析试卷}

\subsection{题目1}
1．设常数 \(\alpha>0\) ，计算积分 \(\int_{1}^{+\infty} \frac{\cos (\ln x)}{x^{\alpha+1}} \mathrm{~d} x\) ．

\subsection{题目2}
2．设 \(L\) 为闭区域 \(D=\left\{(x, y) \mid 0 \leq x \leq 1, x^{3} \leq y \leq \sqrt[3]{x}\right\}\) 的边界，取逆时针方向，计算曲线积分

\[
I=\int_{L}\left(x^{3} y+e^{y}+\ln \left(e^{x}+e^{y}\right)\right) \mathrm{d} x+\left(2 x-2 y+x y^{3}+x e^{y}\right) \mathrm{d} y
\]

\subsection{题目3}
3．设 \(\Omega\) 为 \(\mathbb{R}^{n}\) 中的有界区域，函数 \(f(x)\) 在 \(\bar{\Omega}\) 上连续，对满足 \(U \subseteq \Omega\) 的任意开集 \(U\) ，象集 \(f(U)=\) \(\{f(x) \mid x \in U\}\) 为 \(\mathbb{R}^{1}\) 的开子集，证明：

\[
\max _{x \in \bar{\Omega}} f(x)=\max _{x \in \partial \Omega} f(x) .
\]

\subsection{题目4}
4．令 \(f(x)=\sum_{n=2}^{\infty} \frac{1}{(n \ln n)^{x}}\) ，证明 \(f(x)\) 在 \((1,+\infty)\) 上连续可微．

\subsection{题目5}
5．求函数项级数 \(\sum_{n=1}^{\infty} \frac{\sin n x}{2^{n}}\) 和 \(\sum_{n=1}^{\infty} \frac{1}{n 2^{n}} \sin ^{2} \frac{n x}{2}\) 的和函数．

\subsection{题目6}
6．设数列 \(\left\{\lambda_{n}\right\}\) 满足 \(\lambda_{n} \in\left(\frac{1}{e},+\infty\right), n=1,2, \cdots\) ，且 \(\overline{\lim _{n \rightarrow \infty}} \lambda_{n}>\frac{1}{e}\) ．令

\[
f_{0}(x)=x, f_{n}(x)=\lambda_{n} e^{f_{n-1}(x)}, n=1,2, \cdots .
\]

证明：对任意的 \(x>0\) ，都有 \(\lim _{n \rightarrow \infty} f_{n}(x)=+\infty\) ．

\subsection{题目7}
7．设 \(a_{n}=\int_{\frac{1}{2}}^{2} \frac{2^{x}}{\left(1+x^{2}\right)^{n}} \mathrm{~d} x, n=1,2, \cdots\) ，证明： \(\lim _{n \rightarrow \infty} \frac{a_{n+1}}{a_{n}}=\frac{4}{5}\) ．

\subsection{题目8}
8．证明无穷乘积 \(\prod_{n=1}^{\infty}\left(1+\frac{x}{n^{2}}\right)\) 对于任意的 \(x>0\) 都收敛，并求极限

\[
\lim _{x \rightarrow+\infty} \frac{\ln \left(\prod_{n=1}^{\infty}\left(1+\frac{x}{n^{2}}\right)\right)}{\sqrt{x}}
\]
