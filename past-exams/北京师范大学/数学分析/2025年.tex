\section*{北京师范大学2025年数学分析真题}
\section*{2．北京师范大学2025年数学分析试卷}

\subsection{题目1}
1．（15 分）已知数列 \(\left\{a_{n}\right\}\) 满足 \(a_{1}=\sqrt{3}, a_{n+1}=\sqrt{3+a_{n}}\) ，证明：\(\left\{a_{n}\right\}\) 收敛．

\subsection{题目2}
2．（15 分）求极限 \(\lim _{n \rightarrow \infty} \frac{\pi}{n} \sum_{k=1}^{n} \frac{1}{2+\cos \frac{k \pi}{n}}\) ．

\subsection{题目3}
3．（15 分）已知 \(f(x)\) 在 \([0,2]\) 上可导，且 \(f(0)=0, f^{\prime}(x)\) 在 \([0,2]\) 上连续．证明：

\[
\int_{0}^{2}\left|f(x) f^{\prime}(x)\right| \mathrm{d} x \leq \int_{0}^{2}\left|f^{\prime}(x)\right|^{2} \mathrm{~d} x .
\]

\subsection{题目4}
4．（15 分）求级数和 \(\sum_{n=0}^{\infty} \frac{1}{n!}(n+1)(n+2) x^{n}\) ．

\subsection{题目5}
5．（15 分）已知 \(A \in \mathbb{R}\) ，函数 \(f(x)\) 在 \([0,+\infty)\) 上连续，且 \(\lim _{x \rightarrow+\infty} f(x)=A\) ．证明：

\[
\lim _{x \rightarrow+\infty} \frac{1}{x} \int_{0}^{x} f(t) \mathrm{d} t=A
\]

\subsection{题目6}
6．（15 分）判断 \(\int_{2}^{+\infty} \frac{\sin x}{\ln x} \arctan x \mathrm{~d} x\) 的敛散性，并说明理由．

\subsection{题目7}
7．（15 分）求

\[
f(x, y)=-x^{2} y(x+y-4)
\]

在与 \(x\) 轴，\(y\) 轴，直线 \(y=6-x\) 围成的闭区域 \(D\) 中的极值，最大值与最小值。

\subsection{题目8}
8．（15 分）已知 \(f(x, y)=g(|x y|)\) ，且 \(g(0)=0\) ，在 \(z>0\) 附近，有 \(|g(z)| \leq z^{\alpha}, \alpha>\frac{1}{2}\) ．证明：\(f(x, y)\)在 \((0,0)\) 处可微．

\subsection{题目9}
9．（15 分）已知 \(\sum\) 为单位球面 \(x^{2}+y^{2}+z^{2}=1\) ，取外侧．求曲面积分

\[
\iint_{\Sigma} \frac{x \mathrm{~d} y \mathrm{~d} z+y \mathrm{~d} z \mathrm{~d} x+z \mathrm{~d} x \mathrm{~d} y}{\left(2 x^{2}+2 y^{2}+z^{2}\right)^{\frac{3}{2}}} .
\]

\subsection{题目10}
10．（15 分）求三重积分

\[
\iiint_{x^{2}+y^{2}+z^{2} \leq 2 z}\left(\sqrt{x^{2}+y^{2}+z^{2}}+x-y^{3}\right) \mathrm{d} V
\]
